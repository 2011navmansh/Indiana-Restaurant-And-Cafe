2. Extensions -> Apps Script -> ye code paste karo:

function doPost(e) {
  const sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
  const data = JSON.parse(e.postData.contents);
  sheet.appendRow([data.id||'', data.name||'', data.phone||'', data.date||'', data.guests||'', data.message||'', data.createdAt||'', data.type||'booking', JSON.stringify(data.items||''), data.total||'']);
  return ContentService.createTextOutput(JSON.stringify({ok:true})).setMimeType(ContentService.MimeType.JSON);
}
function doGet(e) {
  const sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
  const rows = sheet.getDataRange().getValues();
  const headers = rows[0];
  const out = rows.slice(1).map(r => { let o={}; headers.forEach((h,i)=>o[h]=r[i]); return o; });
  const type = e.parameter.type;
  const filtered = type ? out.filter(x=> (x.type||'booking')===type) : out;
  return ContentService.createTextOutput(JSON.stringify(filtered)).setMimeType(ContentService.MimeType.JSON);
}
