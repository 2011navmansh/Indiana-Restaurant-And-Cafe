CREATE TABLE IF NOT EXISTS bookings (
  id TEXT PRIMARY KEY, name TEXT NOT NULL, phone TEXT NOT NULL, email TEXT,
  date TEXT NOT NULL, time TEXT NOT NULL, guests INTEGER NOT NULL DEFAULT 2,
  seating TEXT, occasion TEXT, notes TEXT,
  status TEXT NOT NULL DEFAULT 'pending', createdAt TEXT NOT NULL, updatedAt TEXT
);
CREATE INDEX IF NOT EXISTS idx_bookings_date ON bookings(date);
CREATE INDEX IF NOT EXISTS idx_bookings_status ON bookings(status);

CREATE TABLE IF NOT EXISTS orders (
  id TEXT PRIMARY KEY, name TEXT, phone TEXT, items TEXT, total INTEGER DEFAULT 0,
  status TEXT NOT NULL DEFAULT 'new', createdAt TEXT NOT NULL, updatedAt TEXT
);

CREATE TABLE IF NOT EXISTS settings (key TEXT PRIMARY KEY, value TEXT);
INSERT OR IGNORE INTO settings (key, value) VALUES ('door_open', '1');

CREATE TABLE IF NOT EXISTS sessions (

function doPost(e) {
  const sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
  const data = JSON.parse(e.postData.contents);

  if (data.type === 'order') {
    const o = data.order || {};
    sheet.appendRow([o.id||'', 'ORDER', o.name||'', o.phone||'', '', '', '',
                     '', '', JSON.stringify(o.items||[]), o.status||'new', o.createdAt||'']);
  } else {
    const b = data.booking || {};
    sheet.appendRow([b.id||'', 'BOOKING', b.name||'', b.phone||'', b.date||'', b.time||'',
                     b.guests||'', b.seating||'', b.occasion||'', b.notes||'', b.status||'pending', b.createdAt||'']);
  }

  return ContentService.createTextOutput(JSON.stringify({ok:true}))
                       .setMimeType(ContentService.MimeType.JSON);
}

/* ===== INDIANA RESTAURANT & CAFE - Sheet + FREE email automation ===== */
const CONFIG = {
  RESTAURANT: 'Indiana Restaurant & Cafe',

function doPost(e) {
  let data = {};
  try { data = JSON.parse(e.postData.contents); } catch (err) {}
  if (data.type === 'order') return handleOrder(data.order || {});
  if (data.type === 'status') return handleStatus(data.booking || {});
  return handleBooking(data.booking || data);
}
function doGet() { return out({ ok: true, service: 'indiana-webhook' }); }

/* ---- new booking: sheet row + "request received" email ---- */
function handleBooking(b) {
  upsert(b.id, [b.id||'', 'BOOKING', b.name||'', b.phone||'', b.date||'', b.time||'', b.guests||'', b.seating||'', b.occasion||'', b.notes||'', b.status||'pending', b.createdAt||'']);
  if (b.email) {
    sendMail(b.email, 'Booking request received - ' + (b.id || ''),
      shell('Booking Request Received',
        '<p>Hi ' + first(b.name) + ',</p>'
      + '<p>Thank you! Your table request has been received. Our team is checking availability right now - you will get <b>one more email</b> as soon as it is confirmed.</p>'
      + details(b)
      + '<p style="color:#9a9a9a;font-size:12px">Need a change? Reply to this email or call us: ' + CONFIG.PHONE + '</p>'));
  }
  if (CONFIG.NOTIFY_OWNER_ON_NEW && CONFIG.OWNER_EMAIL) {
    sendMail(CONFIG.OWNER_EMAIL, 'New booking - ' + (b.id||'') + ' - ' + (b.date||'') + ' ' + (b.time||''), shell('New Booking Request', details(b)));
  }
  return out({ ok: true });
}

/* ---- owner CONFIRM / CANCEL -> guest email ---- */
function handleStatus(b) {
  upsert(b.id, [b.id||'', 'BOOKING', b.name||'', b.phone||'', b.date||'', b.time||'', b.guests||'', b.seating||'', b.occasion||'', b.notes||'', b.status||'', b.createdAt||'']);
  if (b.status === 'confirmed' && b.email) {
    sendMail(b.email, 'Table confirmed - ' + (b.id || ''),
      shell('Your Table is Confirmed',
        '<p>Hi ' + first(b.name) + ',</p>'
      + '<p>Great news - your table at <b>' + CONFIG.RESTAURANT + '</b> is <b>CONFIRMED</b>. We look forward to hosting you!</p>'
      + details(b)
      + '<p style="color:#9a9a9a;font-size:12px">Please arrive within 20 minutes of your slot. Running late? Call ' + CONFIG.PHONE + '</p>'));
  }
  if (b.status === 'cancelled' && b.email) {
    sendMail(b.email, 'Update on your reservation - ' + (b.id || ''),
      shell('Reservation Update',
        '<p>Hi ' + first(b.name) + ',</p>'
      + '<p>We are sorry - we could not confirm your table for <b>' + (b.date||'') + ' at ' + (b.time||'') + '</b>. That slot may be full.</p>'
      + '<p>We would love to host you another time - book again on our website or call us directly.</p>'
      + details(b)));
  }
  return out({ ok: true });
}

/* ---- orders ---- */
function handleOrder(o) {
  const items = (o.items || []).map(function (i) { return i.qty + 'x ' + i.name; }).join(', ');
  upsert(o.id, [o.id||'', 'ORDER', o.name||'', o.phone||'', '', '', '', '', '', items + ' | Total: Rs ' + (o.total||0), o.status||'new', o.createdAt||'']);
  return out({ ok: true });
}

/* ---- sheet helpers ---- */
function upsert(id, row) {
  const sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
  if (sheet.getLastRow() === 0) {
    sheet.appendRow(['ID','TYPE','NAME','PHONE','DATE','TIME','GUESTS','SEATING','OCCASION','NOTES','STATUS','CREATED']);
  }
  const at = findRow(sheet, id);
  if (at) sheet.getRange(at, 1, 1, row.length).setValues([row]);
  else sheet.appendRow(row);
}
function findRow(sheet, id) {
  if (!id || sheet.getLastRow() < 2) return 0;
  const ids = sheet.getRange(2, 1, sheet.getLastRow() - 1, 1).getValues();
  for (let i = 0; i < ids.length; i++) if (String(ids[i][0]) === String(id)) return i + 2;
  return 0;
}

/* ---- email helpers ---- */
function sendMail(to, subject, html) {
  try { if (to && to.indexOf('@') > 0) MailApp.sendEmail({ to: to, subject: subject, htmlBody: html }); }
  catch (err) { Logger.log('mail error: ' + err); }
}
function first(n) { return String(n || '').split(' ')[0] || 'there'; }
function details(b) {
  const rows = [['Reference', b.id], ['Name', b.name], ['Date', b.date], ['Time', b.time], ['Guests', b.guests], ['Seating', b.seating || 'Any'], ['Occasion', b.occasion || '-']];
  return '<table style="width:100%;border-collapse:collapse;margin:14px 0">'
    + rows.map(function (r) {
        return '<tr><td style="padding:7px 0;color:#9a9a9a;font-size:11px;letter-spacing:1px;text-transform:uppercase;width:38%">' + r[0] + '</td>'
          + '<td style="padding:7px 0;color:#ffffff;font-weight:bold">' + ((r[1] === 0 || r[1]) ? r[1] : '-') + '</td></tr>';
      }).join('')
    + '</table>';
}
function shell(title, inner) {

    + '<div style="font-size:12px;letter-spacing:2px;font-weight:bold">' + String(CONFIG.RESTAURANT).toUpperCase() + '</div>'
    + '<div style="font-size:21px;font-weight:bold;margin-top:4px">' + title + '</div></div>'

    + '</div></div>';
}
